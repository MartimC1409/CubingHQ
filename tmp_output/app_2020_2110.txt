L2070|    function generateClockScramble() {

L2072|        //   Side 1: 9 pins (UR, DR, DL, UL, U, R, D, L, ALL), each with ±1-5
L2073|        //   y2 rotation
L2074|        //   Side 2: 5 pins (U, D, L, R, ALL) - the 4 corner pins (UR/DR/DL/UL) are pushed up
L2075|        //   Optional: 0+ pins lifted at the end (UR DR DL UL)

L2078|        const moves = [];
L2079|        const randPin = () => {
L2080|            // ±1-5 with balanced sign distribution
L2081|            const sign = Math.random() < 0.5 ? -1 : 1;
L2082|            const mag = 1 + Math.floor(Math.random() * 5); // 1..5
L2083|            return sign * mag;
L2084|        };
L2085|        side1Pins.forEach(pin => {
L2086|            const v = randPin();
L2087|            moves.push(`${pin}${v > 0 ? v + '+' : Math.abs(v) + '-'}`);

L2093|        });
L2094|        // 0+ pins lifted (UR DR DL UL) - WCA spec: pick 0, 1, or 2 pins; each +1..+5
L2095|        const liftPins = ['UR', 'DR', 'DL', 'UL'];
L2096|        const liftCount = Math.random() < 0.33 ? (Math.random() < 0.5 ? 1 : 2) : 0;
L2097|        const lifted = [];

L2101|                const j = Math.floor(Math.random() * (i + 1));
L2102|                [pool[i], pool[j]] = [pool[j], pool[i]];
L2103|            }
L2104|            for (let i = 0; i < liftCount; i++) {
L2105|                const v = 1 + Math.floor(Math.random() * 5);
L2106|                lifted.push(`${pool[i]}${v}+`);
L2107|            }
L2108|        }
L2109|        if (lifted.length) moves.push(lifted.join(' '));
